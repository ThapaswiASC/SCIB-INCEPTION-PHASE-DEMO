### 3.1 KanbanBoardComponent (Feature)

**Purpose:** Primary container for kanban board with three-column layout

**TypeScript Implementation:**

```typescript
// kanban-board.component.ts
import { Component, OnInit, Input, Output, EventEmitter, ChangeDetectionStrategy } from '@angular/core';
import { CdkDragDrop, moveItemInArray, transferArrayItem } from '@angular/cdk/drag-drop';
import { Task, KanbanColumn, CreateTaskRequest } from '../../models/task.model';
import { TaskService } from '../../services/task.service';
import { Observable, BehaviorSubject } from 'rxjs';

@Component({
  selector: 'app-kanban-board',
  templateUrl: './kanban-board.component.html',
  styleUrls: ['./kanban-board.component.scss'],
  changeDetection: ChangeDetectionStrategy.OnPush
})
export class KanbanBoardComponent implements OnInit {
  @Input() loading: boolean = false;
  @Input() error: string | null = null;
  @Output() taskMove = new EventEmitter<{taskId: string, newColumnId: string}>();
  @Output() taskSelect = new EventEmitter<Task>();
  @Output() addTask = new EventEmitter<void>();

  columns: KanbanColumn[] = [
    { id: 'todo', title: 'To Do', tasks: [] },
    { id: 'inprogress', title: 'In Progress', tasks: [] },
    { id: 'done', title: 'Done', tasks: [] }
  ];

  tasks$ = new BehaviorSubject<Task[]>([]);
  selectedTask: Task | null = null;
  draggedTask: Task | null = null;

  constructor(private taskService: TaskService) {}

  ngOnInit(): void {
    this.loadTasks();
  }

  loadTasks(): void {
    this.taskService.getTasks().subscribe({
      next: (tasks) => {
        this.tasks$.next(tasks);
        this.distributeTasksToColumns(tasks);
      },
      error: (error) => {
        console.error('Error loading tasks:', error);
      }
    });
  }

  distributeTasksToColumns(tasks: Task[]): void {
    this.columns.forEach(column => {
      column.tasks = tasks.filter(task => task.status === column.id);
    });
  }

  onTaskDrop(event: CdkDragDrop<Task[]>): void {
    if (event.previousContainer === event.container) {
      moveItemInArray(event.container.data, event.previousIndex, event.currentIndex);
    } else {
      const task = event.previousContainer.data[event.previousIndex];
      const newColumnId = this.getColumnIdFromContainer(event.container);
      
      transferArrayItem(
        event.previousContainer.data,
        event.container.data,
        event.previousIndex,
        event.currentIndex
      );

      this.taskMove.emit({ taskId: task.id, newColumnId });
      this.updateTaskStatus(task.id, newColumnId);
    }
  }

  private getColumnIdFromContainer(container: any): string {
    return container.id;
  }

  private updateTaskStatus(taskId: string, newStatus: string): void {
    this.taskService.updateTask(taskId, { status: newStatus }).subscribe({
      next: () => {
        console.log('Task status updated successfully');
      },
      error: (error) => {
        console.error('Error updating task status:', error);
        // Revert the UI change on error
        this.loadTasks();
      }
    });
  }

  onTaskSelect(task: Task): void {
    this.selectedTask = task;
    this.taskSelect.emit(task);
  }

  onAddTaskClick(): void {
    this.addTask.emit();
  }

  trackByTaskId(index: number, task: Task): string {
    return task.id;
  }

  trackByColumnId(index: number, column: KanbanColumn): string {
    return column.id;
  }
}
```

### 3.2 KanbanColumnComponent (Feature)

**Purpose:** Individual column container with drag and drop support

**TypeScript Implementation:**

```typescript
// kanban-column.component.ts
import { Component, Input, Output, EventEmitter, ChangeDetectionStrategy } from '@angular/core';
import { CdkDragDrop } from '@angular/cdk/drag-drop';
import { Task, KanbanColumn } from '../../models/task.model';

@Component({
  selector: 'app-kanban-column',
  templateUrl: './kanban-column.component.html',
  styleUrls: ['./kanban-column.component.scss'],
  changeDetection: ChangeDetectionStrategy.OnPush
})
export class KanbanColumnComponent {
  @Input() column!: KanbanColumn;
  @Input() tasks: Task[] = [];
  @Input() allowDrop: boolean = true;
  @Output() taskDrop = new EventEmitter<CdkDragDrop<Task[]>>();
  @Output() taskSelect = new EventEmitter<Task>();

  isDropTarget: boolean = false;

  onTaskDrop(event: CdkDragDrop<Task[]>): void {
    this.taskDrop.emit(event);
    this.isDropTarget = false;
  }

  onTaskSelect(task: Task): void {
    this.taskSelect.emit(task);
  }

  onDragEntered(): void {
    this.isDropTarget = true;
  }

  onDragExited(): void {
    this.isDropTarget = false;
  }

  trackByTaskId(index: number, task: Task): string {
    return task.id;
  }
}
```

```typescript
// task-card.component.ts
import { Component, Input, Output, EventEmitter, ChangeDetectionStrategy } from '@angular/core';
import { Task, TaskPriority } from '../../models/task.model';

@Component({
  selector: 'app-task-card',
  templateUrl: './task-card.component.html',
  styleUrls: ['./task-card.component.scss'],
  changeDetection: ChangeDetectionStrategy.OnPush
})
export class TaskCardComponent {
  @Input() task!: Task;
  @Input() draggable: boolean = true;
  @Input() selected: boolean = false;
  @Output() select = new EventEmitter<Task>();
  @Output() dragStart = new EventEmitter<Task>();
  @Output() dragEnd = new EventEmitter<void>();

  isDragging: boolean = false;

  onCardClick(event: Event): void {
    event.stopPropagation();
    this.select.emit(this.task);
  }

  onDragStart(): void {
    this.isDragging = true;
    this.dragStart.emit(this.task);
  }

  onDragEnd(): void {
    this.isDragging = false;
    this.dragEnd.emit();
  }

  getPriorityClass(): string {
    switch (this.task.priority) {
      case TaskPriority.HIGH:
        return 'priority-high';
      case TaskPriority.MEDIUM:
        return 'priority-medium';
      case TaskPriority.LOW:
        return 'priority-low';
      default:
        return 'priority-medium';
    }
  }

  getPriorityLabel(): string {
    switch (this.task.priority) {
      case TaskPriority.HIGH:
        return 'High Priority';
      case TaskPriority.MEDIUM:
        return 'Medium Priority';
      case TaskPriority.LOW:
        return 'Low Priority';
      default:
        return 'Medium Priority';
    }
  }

  getStatusColor(): string {
    switch (this.task.status) {
      case 'todo':
        return '#f39c12';
      case 'inprogress':
        return '#3498db';
      case 'done':
        return '#27ae60';
      default:
        return '#95a5a6';
    }
  }

  formatDate(date: Date): string {
    return new Intl.DateTimeFormat('en-US', {
      month: 'short',
      day: 'numeric'
    }).format(new Date(date));
  }

  isOverdue(): boolean {
    if (!this.task.dueDate) return false;
    return new Date(this.task.dueDate) < new Date() && this.task.status !== 'done';
  }
}
```

```typescript
// sidebar.component.ts
import { Component, Input, Output, EventEmitter, ChangeDetectionStrategy } from '@angular/core';
import { Router } from '@angular/router';
import { User } from '../../models/user.model';

interface NavigationItem {
  id: string;
  label: string;
  icon: string;
  route: string;
  children?: NavigationItem[];
}

@Component({
  selector: 'app-sidebar',
  templateUrl: './sidebar.component.html',
  styleUrls: ['./sidebar.component.scss'],
  changeDetection: ChangeDetectionStrategy.OnPush
})
export class SidebarComponent {
  @Input() activeRoute: string = '';
  @Input() user!: User;
  @Output() navigate = new EventEmitter<string>();

  expandedSections: string[] = ['task-management'];

  navigationItems: NavigationItem[] = [
    {
      id: 'task-management',
      label: 'Task Management',
      icon: '📋',
      route: '/tasks',
      children: [
        { id: 'kanban', label: 'Kanban Board', icon: '📊', route: '/kanban' },
        { id: 'collaborative', label: 'Collaborative Board', icon: '👥', route: '/collaborative' },
        { id: 'my-tasks', label: 'My Tasks', icon: '✅', route: '/my-tasks' }
      ]
    },
    {
      id: 'analytics',
      label: 'Analytics',
      icon: '📈',
      route: '/analytics',
      children: [
        { id: 'overview', label: 'Overview', icon: '📊', route: '/analytics/overview' },
        { id: 'team-performance', label: 'Team Performance', icon: '👥', route: '/analytics/team' },
        { id: 'reports', label: 'Reports', icon: '📄', route: '/analytics/reports' }
      ]
    },
    {
      id: 'configuration',
      label: 'Configuration',
      icon: '⚙️',
      route: '/config',
      children: [
        { id: 'workflow', label: 'Workflow Rules', icon: '🔄', route: '/config/workflow' },
        { id: 'team', label: 'Team Settings', icon: '👥', route: '/config/team' },
        { id: 'integrations', label: 'Integrations', icon: '🔗', route: '/config/integrations' }
      ]
    }
  ];

  constructor(private router: Router) {}

  onNavigate(route: string): void {
    this.navigate.emit(route);
    this.router.navigate([route]);
  }

  toggleSection(sectionId: string): void {
    const index = this.expandedSections.indexOf(sectionId);
    if (index > -1) {
      this.expandedSections.splice(index, 1);
    } else {
      this.expandedSections.push(sectionId);
    }
  }

  isSectionExpanded(sectionId: string): boolean {
    return this.expandedSections.includes(sectionId);
  }

  isRouteActive(route: string): boolean {
    return this.activeRoute === route || this.activeRoute.startsWith(route + '/');
  }

  getUserInitials(): string {
    if (!this.user?.name) return 'U';
    return this.user.name
      .split(' ')
      .map(name => name.charAt(0))
      .join('')
      .toUpperCase()
      .slice(0, 2);
  }
}
```

**TypeScript Implementation:**

```typescript
// header.component.ts
import { Component, Input, Output, EventEmitter, ChangeDetectionStrategy } from '@angular/core';
import { User } from '../../models/user.model';
import { debounceTime, distinctUntilChanged, Subject } from 'rxjs';

@Component({
  selector: 'app-header',
  templateUrl: './header.component.html',
  styleUrls: ['./header.component.scss'],
  changeDetection: ChangeDetectionStrategy.OnPush
})
export class HeaderComponent {
  @Input() user!: User;
  @Input() searchQuery: string = '';
  @Output() search = new EventEmitter<string>();
  @Output() notifications = new EventEmitter<void>();
  @Output() settings = new EventEmitter<void>();
  @Output() profile = new EventEmitter<void>();

  isSearchFocused: boolean = false;
  searchSubject = new Subject<string>();
  notificationCount: number = 3;
  showNotificationBadge: boolean = true;

  constructor() {
    // Debounce search input
    this.searchSubject.pipe(
      debounceTime(300),
      distinctUntilChanged()
    ).subscribe(query => {
      this.search.emit(query);
    });
  }

  onSearchInput(event: Event): void {
    const target = event.target as HTMLInputElement;
    this.searchSubject.next(target.value);
  }

  onSearchFocus(): void {
    this.isSearchFocused = true;
  }

  onSearchBlur(): void {
    this.isSearchFocused = false;
  }

  onNotificationsClick(): void {
    this.showNotificationBadge = false;
    this.notifications.emit();
  }

  onSettingsClick(): void {
    this.settings.emit();
  }

  onProfileClick(): void {
    this.profile.emit();
  }

  getUserInitials(): string {
    if (!this.user?.name) return 'U';
    return this.user.name
      .split(' ')
      .map(name => name.charAt(0))
      .join('')
      .toUpperCase()
      .slice(0, 2);
  }

  clearSearch(): void {
    this.searchSubject.next('');
  }
}
```

```typescript
    this.taskForm = this.formBuilder.group({
      title: ['', [Validators.required, Validators.minLength(3), Validators.maxLength(100)]],
      description: ['', [Validators.maxLength(500)]],
      priority: [TaskPriority.MEDIUM, [Validators.required]],
      status: ['todo', [Validators.required]],
      assigneeId: [''],
      dueDate: [''],
      tags: [''],
      estimatedHours: [null, [Validators.min(0.5), Validators.max(100)]]
    });
  }

  onSubmit(): void {
    if (this.taskForm.valid && !this.isSubmitting) {
      this.isSubmitting = true;
      this.validationErrors = [];

      const formValue = this.taskForm.value;
      const createTaskRequest: CreateTaskRequest = {
        title: formValue.title.trim(),
        description: formValue.description?.trim() || '',
        priority: formValue.priority,
        status: formValue.status,
        assigneeId: formValue.assigneeId || null,
        dueDate: formValue.dueDate ? new Date(formValue.dueDate) : null,
        tags: this.parseTags(formValue.tags),
        estimatedHours: formValue.estimatedHours || null
      };

      this.submit.emit(createTaskRequest);
    } else {
      this.validateForm();
    }
  }

  onClose(): void {
    if (!this.isSubmitting) {
      this.resetForm();
      this.close.emit();
    }
  }

  onBackdropClick(event: Event): void {
    if (event.target === event.currentTarget) {
      this.onClose();
    }
  }

  resetForm(): void {
    this.taskForm.reset({
      priority: TaskPriority.MEDIUM,
      status: 'todo'
    });
    this.validationErrors = [];
    this.isSubmitting = false;
  }

  validateForm(): void {
    this.validationErrors = [];
    
    Object.keys(this.taskForm.controls).forEach(key => {
      const control = this.taskForm.get(key);
      if (control && control.invalid && control.touched) {
        const errors = control.errors;
        if (errors) {
          if (errors['required']) {
            this.validationErrors.push({
              field: key,
              message: `${this.getFieldLabel(key)} is required`
            });
          }
          if (errors['minlength']) {
            this.validationErrors.push({
              field: key,
              message: `${this.getFieldLabel(key)} must be at least ${errors['minlength'].requiredLength} characters`
            });
          }
          if (errors['maxlength']) {
            this.validationErrors.push({
              field: key,
              message: `${this.getFieldLabel(key)} cannot exceed ${errors['maxlength'].requiredLength} characters`
            });
          }
          if (errors['min']) {
            this.validationErrors.push({
              field: key,
              message: `${this.getFieldLabel(key)} must be at least ${errors['min'].min}`
            });
          }
          if (errors['max']) {
            this.validationErrors.push({
              field: key,
              message: `${this.getFieldLabel(key)} cannot exceed ${errors['max'].max}`
            });
          }
        }
      }
    });
  }

  getFieldLabel(fieldName: string): string {
    const labels: { [key: string]: string } = {
      title: 'Title',
      description: 'Description',
      priority: 'Priority',
      status: 'Status',
      assigneeId: 'Assignee',
      dueDate: 'Due Date',
      tags: 'Tags',
      estimatedHours: 'Estimated Hours'
    };
    return labels[fieldName] || fieldName;
  }

  parseTags(tagsString: string): string[] {
    if (!tagsString) return [];
    return tagsString
      .split(',')
      .map(tag => tag.trim())
      .filter(tag => tag.length > 0)
      .slice(0, 5); // Limit to 5 tags
  }

  getAssigneeById(id: string): User | undefined {
    return this.users.find(user => user.id === id);
  }

  getPriorityConfig(priority: TaskPriority) {
    return this.taskPriorities.find(p => p.value === priority);
  }

  onFormSubmissionComplete(): void {
    this.isSubmitting = false;
    this.resetForm();
    this.close.emit();
  }

  onFormSubmissionError(error: any): void {
    this.isSubmitting = false;
    console.error('Task creation error:', error);
    // Handle API errors here
  }

  markAllFieldsAsTouched(): void {
    Object.keys(this.taskForm.controls).forEach(key => {
      this.taskForm.get(key)?.markAsTouched();
    });
  }

  isFieldInvalid(fieldName: string): boolean {
    const field = this.taskForm.get(fieldName);
    return !!(field && field.invalid && field.touched);
  }

  getFieldError(fieldName: string): string {
    const error = this.validationErrors.find(err => err.field === fieldName);
    return error ? error.message : '';
  }
}
```
